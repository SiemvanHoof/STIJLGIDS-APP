  --kleur-achtergrond: #F3EEE7;

  --tekst: var(--kleur-tekst);
  --hoofd: var(--kleur-hoofdkleur);
  --tweede: var(--kleur-accent);
  --op-hoofd: #FFFFFF;
  --op-tweede: #1C1C1C;
  --gedempt: #72706D;
  --lijn: #D3CFC9;

  /* Knoppen */

}
